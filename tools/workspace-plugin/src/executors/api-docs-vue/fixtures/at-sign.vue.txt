  /**
   * The content of the thing. Use @click to listen. When not set, the thing is empty.

   * @example

  default?(): unknown;
}>();

  /**
   * Moves focus into the thing. Use @click to listen.
   *
   * @example
   * <Thing @click="save">Save</Thing>
   */
  focus(): void {
    return undefined;
  },
});
</script>
